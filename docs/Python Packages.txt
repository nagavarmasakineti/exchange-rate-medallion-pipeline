import requests

### GET: Retrieves Data --200
def fetch_api_data(url):
  response = requests.get(url)

### POST: Create New Data --201
def create_api_data(url, payload):
  response - requests.post(url, json=payload)

### PUT: Update Exisiting Data --200
def update_api_data(url, payload):
  response - requests.put(url, json=payload)

### DELETE: Remove Data --204 No content
def delete_api_data(url):
  response - requests.delete(url)

### READ/SELECT : Fetch records as list of tuples
def fetch_db_records(cursor, table_name, condition="1=1"):
  query = f"SELECT * FROM  {table_name} WHERE {condition};"
  cursor.execute(query)
  return cursor.fetchall()

### UPDATE : Update exisiting database records
def update_db_record(cursor, table_name, set_clause, condition):
  query = f"UPDATE {table_name} SET {set_clause} WHERE {condition};"
  cursor.execute(query)

### DELETE : Remove database records
def delete_db_record(cursor, table_name, condition):
  query = f"DELETE FROM {table_name} WHERE {condition};"
  cursor.execute(query)
